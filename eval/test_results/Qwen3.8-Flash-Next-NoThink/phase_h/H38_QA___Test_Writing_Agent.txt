def is_valid_password(password):
    # Check length
    if len(password) < 8 or len(password) > 64:
        return False
    
    # Check for spaces
    if ' ' in password:
        return False
    
    # Check for at least 1 uppercase letter
    has_upper = False
    for c in password:
        if c.isupper():
            has_upper = True
            break
    if not has_upper:
        return False
    
    # Check for at least 1 lowercase letter
    has_lower = False
    for c in password:
        if c.islower():
            has_lower = True
            break
    if not has_lower:
        return False
    
    # Check for at least 1 digit
    has_digit = False
    for c in password:
        if c.isdigit():
            has_digit = True
            break
    if not has_digit:
        return False
    
    # Check for at least 1 special character from: !@#$%^&*()-_+=
    special_chars = "!@#$%^&*()-_+="
    has_special = False
    for c in password:
        if c in special_chars:
            has_special = True
            break
    if not has_special:
        return False
    
    # Cannot start with a digit
    if password[0].isdigit():
        return False
    
    # Cannot contain more than 3 consecutive identical characters
    count = 1
    for i in range(1, len(password)):
        if password[i] == password[i-1]:
            count += 1
            if count > 3:
                return False
        else:
            count = 1
    
    # Cannot contain the word "password" (case-insensitive)
    if "password" in password.lower():
        return False
    
    return True

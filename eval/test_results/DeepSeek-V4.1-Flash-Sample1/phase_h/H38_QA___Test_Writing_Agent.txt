def is_valid_password(password):
    if not isinstance(password, str):
        return False
    if len(password) < 8 or len(password) > 64:
        return False
    if password[0].isdigit():
        return False
    if any(c.isspace() for c in password):
        return False
    if "password" in password.lower():
        return False
    if not any(c.isupper() for c in password):
        return False
    if not any(c.islower() for c in password):
        return False
    if not any(c.isdigit() for c in password):
        return False

    specials = "!@#$%^&*()-_+="
    if not any(c in specials for c in password):
        return False

    count = 1
    for i in range(1, len(password)):
        if password[i] == password[i - 1]:
            count += 1
            if count > 3:
                return False
        else:
            count = 1

    return True


def run_tests():
    tests = [
        ("Abcdef1!", True),
        ("Abcdefg1", False),
        ("abcdef1!", False),
        ("ABCDEF1!", False),
        ("Abcdefg!", False),
        ("Abcdef1! ", False),
        ("1Abcdef!", False),
        ("Abc1111!", False),
        ("Abcd111!", True),
        ("Password1!", False),
        ("Passw0rd1!", True),
        ("Abcdef1!@#$%^&*()-_+=", True),
        ("Abcdef1!" + "aB" * 29, False),
        ("Abcdef1", False),
        ("Abcdef1!@", True),
    ]
    return [(p, expected, is_valid_password(p)) for p, expected in tests]

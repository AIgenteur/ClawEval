class JSONValidator:
    _ALLOWED_TYPES = ("str", "int", "float", "bool", "list", "dict")

    @staticmethod
    def validate(data, schema):
        errors = []

        if not isinstance(data, dict):
            errors.append(f"data: expected dict, got {JSONValidator._type_name(data)}")
            return {"valid": False, "errors": errors}

        if not isinstance(schema, dict):
            errors.append(f"schema: expected dict, got {JSONValidator._type_name(schema)}")
            return {"valid": False, "errors": errors}

        for field, rules in schema.items():
            if not isinstance(rules, dict):
                errors.append(f"{field}: schema rules must be a dict")
                continue

            required = rules.get("required", False)

            if field not in data:
                if required:
                    errors.append(f"{field}: required field missing")
                continue

            value = data[field]
            expected_type = rules.get("type")

            check_numeric = False
            check_length = False

            if expected_type is not None:
                if expected_type not in JSONValidator._ALLOWED_TYPES:
                    errors.append(f"{field}: unknown type {expected_type}")
                    continue

                if not JSONValidator._matches_type(value, expected_type):
                    errors.append(
                        f"{field}: expected type {expected_type}, "
                        f"got {JSONValidator._type_name(value)}"
                    )
                    continue

                check_numeric = expected_type in ("int", "float")
                check_length = expected_type in ("str", "list")
            else:
                check_numeric = JSONValidator._is_number(value)
                check_length = isinstance(value, (str, list))

            if check_numeric:
                if "min" in rules:
                    min_value = rules["min"]
                    if not JSONValidator._is_number(min_value):
                        errors.append(f"{field}: min must be a number")
                    elif value < min_value:
                        errors.append(f"{field}: value must be >= min {min_value}")

                if "max" in rules:
                    max_value = rules["max"]
                    if not JSONValidator._is_number(max_value):
                        errors.append(f"{field}: max must be a number")
                    elif value > max_value:
                        errors.append(f"{field}: value must be <= max {max_value}")

            if check_length:
                if "min_length" in rules:
                    min_length = rules["min_length"]
                    if not JSONValidator._is_int(min_length):
                        errors.append(f"{field}: min_length must be an integer")
                    elif len(value) < min_length:
                        errors.append(f"{field}: length must be >= min_length {min_length}")

                if "max_length" in rules:
                    max_length = rules["max_length"]
                    if not JSONValidator._is_int(max_length):
                        errors.append(f"{field}: max_length must be an integer")
                    elif len(value) > max_length:
                        errors.append(f"{field}: length must be <= max_length {max_length}")

        return {"valid": not errors, "errors": errors}

    @staticmethod
    def _matches_type(value, expected_type):
        if expected_type == "str":
            return isinstance(value, str)

        if expected_type == "int":
            return isinstance(value, int) and not isinstance(value, bool)

        if expected_type == "float":
            # JSON numbers may be represented as int or float in Python.
            return (
                isinstance(value, float)
                or (isinstance(value, int) and not isinstance(value, bool))
            )

        if expected_type == "bool":
            return isinstance(value, bool)

        if expected_type == "list":
            return isinstance(value, list)

        if expected_type == "dict":
            return isinstance(value, dict)

        return False

    @staticmethod
    def _type_name(value):
        if isinstance(value, bool):
            return "bool"
        if isinstance(value, int):
            return "int"
        if isinstance(value, float):
            return "float"
        if isinstance(value, str):
            return "str"
        if isinstance(value, list):
            return "list"
        if isinstance(value, dict):
            return "dict"
        return type(value).__name__

    @staticmethod
    def _is_number(value):
        return isinstance(value, (int, float)) and not isinstance(value, bool)

    @staticmethod
    def _is_int(value):
        if isinstance(value, bool):
            return False
        if isinstance(value, int):
            return True
        if isinstance(value, float):
            return value.is_integer()
        return False

def is_valid_password(password):
    if not isinstance(password, str):
        return False
    # Length: at least 8, at most 64
    if len(password) < 8 or len(password) > 64:
        return False
    # At least one uppercase letter
    if not any(c.isupper() for c in password):
        return False
    # At least one lowercase letter
    if not any(c.islower() for c in password):
        return False
    # At least one digit
    if not any(c.isdigit() for c in password):
        return False
    # At least one special character from the allowed set
    special = "!@#$%^&*()-_+="
    if not any(c in special for c in password):
        return False
    # No spaces allowed
    if " " in password:
        return False
    # Cannot start with a digit
    if password[0].isdigit():
        return False
    # No more than 3 consecutive identical characters (4+ in a row is invalid)
    for i in range(len(password) - 3):
        if password[i] == password[i + 1] == password[i + 2] == password[i + 3]:
            return False
    # Cannot contain the word "password" (case-insensitive)
    if "password" in password.lower():
        return False
    return True


def run_tests():
    tests = [
        ("Abcdef1!", True),             # valid: meets all rules
        ("Ab1!", False),                # too short (< 8)
        ("Abcdef1!" * 8 + "X", False),  # too long (65 > 64)
        ("abcdef1!", False),            # no uppercase
        ("ABCDEF1!", False),            # no lowercase
        ("Abcdefg!", False),            # no digit
        ("Abcdefg1", False),            # no special character
        ("Abcd ef1!", False),           # contains a space
        ("1Abcdef!", False),            # starts with a digit
        ("Abcccc1!", False),            # 4 consecutive identical chars
        ("Abcccd1!", True),             # 3 consecutive identical chars is OK
        ("MyPassword1!", False),        # contains "password"
        ("MyPASSWORD1!", False),        # contains "PASSWORD" (case-insensitive)
        ("Abcdef1!" * 8, True),         # exactly 64 chars (boundary, valid)
        ("Ab1!xyzW", True),             # exactly 8 chars (boundary, valid)
    ]
    return [(pwd, expected, is_valid_password(pwd)) for pwd, expected in tests]
